// Incluye auditoría segura, mensajería, colaboración CRDT, presencia y aportes.
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function ownsDocument(uid) {
      return signedIn() && request.auth.uid == uid;
    }

    function isActiveStudent(uid) {
      return verifiedEmail() &&
        ownsDocument(uid) &&
        exists(/databases/$(database)/documents/estudiantes/$(uid)) &&
        get(
          /databases/$(database)/documents/estudiantes/$(uid)
        ).data.get('estadoCuenta', 'pendiente') == 'activo';
    }

    function verifiedEmail() {
      return signedIn() &&
        request.auth.token.get('email_verified', false) == true;
    }

    function authenticatedEmail() {
      return request.auth.token.get('email', '');
    }

    function isPrincipalTeacherAdmin() {
      return verifiedEmail() &&
        authenticatedEmail() == 'ipem146centenario@gmail.com';
    }

    function isManagedTeacher() {
      return verifiedEmail() &&
        exists(
          /databases/$(database)/documents/docentesAutorizados/$(authenticatedEmail())
        ) &&
        get(
          /databases/$(database)/documents/docentesAutorizados/$(authenticatedEmail())
        ).data.get('email', '') == authenticatedEmail() &&
        get(
          /databases/$(database)/documents/docentesAutorizados/$(authenticatedEmail())
        ).data.get('activo', false) == true;
    }

    function isTeacher() {
      return isPrincipalTeacherAdmin() || isManagedTeacher();
    }

    function validTeacherRecord(docenteId) {
      return request.resource.data.keys().hasAll([
          'email',
          'nombre',
          'rol',
          'activo',
          'creadoEn',
          'creadoPor',
          'actualizadoEn',
          'actualizadoPor',
          'bajaEn',
          'bajaPor',
          'motivoBaja'
        ]) &&
        request.resource.data.keys().hasOnly([
          'email',
          'nombre',
          'rol',
          'activo',
          'creadoEn',
          'creadoPor',
          'actualizadoEn',
          'actualizadoPor',
          'bajaEn',
          'bajaPor',
          'motivoBaja'
        ]) &&
        request.resource.data.email == docenteId &&
        request.resource.data.email is string &&
        request.resource.data.email.size() >= 5 &&
        request.resource.data.email.size() <= 254 &&
        request.resource.data.nombre is string &&
        request.resource.data.nombre.size() >= 1 &&
        request.resource.data.nombre.size() <= 120 &&
        request.resource.data.rol is string &&
        request.resource.data.activo is bool &&
        request.resource.data.creadoEn is timestamp &&
        request.resource.data.creadoPor is string &&
        request.resource.data.actualizadoEn == request.time &&
        request.resource.data.actualizadoPor == authenticatedEmail() &&
        (
          request.resource.data.bajaEn == null ||
          request.resource.data.bajaEn is timestamp
        ) &&
        request.resource.data.bajaPor is string &&
        request.resource.data.motivoBaja is string &&
        request.resource.data.motivoBaja.size() <= 300 &&
        (
          docenteId == 'ipem146centenario@gmail.com'
            ? (
              request.resource.data.rol == 'administrador' &&
              request.resource.data.activo == true
            )
            : request.resource.data.rol == 'docente'
        );
    }

    function studentMutableFields() {
      return [
        'estudiante',
        'salidasPestana',
        'eventosSalidasPestana',
        'pantallaBloqueada',
        'pantallaBloqueadaEn',
        'pantallaBloqueadaSalidas',
        'pantallaBloqueadaSeccion',
        'codigos',
        'tiemposRestantes',
        'finalizadas',
        'historialResultados',
        'chatIA',
        'ayudasComprension',
        'contadorPrevisualizaciones',
        'estadosRecomendacionesInforme',
        'seccionActiva',
        'versionCodigo',
        'versionApp',
        'nombreGoogle',
        'fotoGoogle',
        'emailVerificado',
        'seguimientoExtension',
        'jitsiParticipacion',
        'versionScript',
        'intentosPortapapeles',
        'presenciaEnLinea',
        'presenciaActualizadaEn',
        'presenciaActualizadaEnCliente',
        'versionLocal',
        'versionLocalEn',
        'actualizadoEn'
      ];
    }

    function teacherMutableStudentFields() {
      return [
        'estadoCuenta',
        'bajaMotivo',
        'bajaFecha',
        'bajaPor',
        'rechazoMotivo',
        'rechazadoEn',
        'rechazadoPor',
        'aprobadoEn',
        'aprobadoPor',
        'revisionSalidas',
        'notasDesafiosDocente',
        'pantallaBloqueada',
        'pantallaBloqueadaEn',
        'pantallaBloqueadaSalidas',
        'pantallaBloqueadaSeccion',
        'bloqueoManual',
        'ultimoBloqueoManual',
        'cantidadDesbloqueos',
        'ultimoDesbloqueo',
        'desbloqueoPantalla',
        'jitsiSala',
        'jitsiGrupal',
        'planesRefuerzo',
        'mensajeDocenteActual',
        'actualizadoEn'
      ];
    }

    function validTeacherStudentUpdate(uid) {
      return isTeacher() &&
        request.resource.data.uid == resource.data.uid &&
        request.resource.data.email == resource.data.email &&
        request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(teacherMutableStudentFields()) &&
        request.resource.data.get('actualizadoEn', null) == request.time &&
        request.resource.data.get('estadoCuenta', resource.data.get('estadoCuenta', 'pendiente')) in [
          'pendiente',
          'activo',
          'rechazado',
          'inactivo'
        ] &&
        request.resource.data.get('bajaMotivo', '') is string &&
        request.resource.data.get('bajaMotivo', '').size() <= 500 &&
        request.resource.data.get('rechazoMotivo', '') is string &&
        request.resource.data.get('rechazoMotivo', '').size() <= 500 &&
        request.resource.data.get('bajaPor', '') is string &&
        request.resource.data.get('bajaPor', '').size() <= 254 &&
        request.resource.data.get('rechazadoPor', '') is string &&
        request.resource.data.get('rechazadoPor', '').size() <= 254 &&
        request.resource.data.get('aprobadoPor', '') is string &&
        request.resource.data.get('aprobadoPor', '').size() <= 254 &&
        request.resource.data.get('revisionSalidas', {}) is map &&
        request.resource.data.get('notasDesafiosDocente', {}) is map &&
        request.resource.data.get('planesRefuerzo', []) is list &&
        request.resource.data.get('planesRefuerzo', []).size() <= 100 &&
        request.resource.data.get('mensajeDocenteActual', {}) is map;
    }

      return [

    function validNewStudent(uid) {
      return ownsDocument(uid) &&
        request.resource.data.keys().hasAll([
          'uid',
          'email',
          'estudiante',
          'estadoCuenta'
        ]) &&
        request.resource.data.keys().hasOnly(newStudentAllowedFields()) &&
        request.resource.data.uid == uid &&
        request.resource.data.email ==
          request.auth.token.get('email', '') &&
        request.resource.data.get('emailVerificado', false) ==
          request.auth.token.get('email_verified', false) &&
        request.resource.data.get('versionLocal', 0) is number &&
        request.resource.data.get('versionLocal', 0) >= 0 &&
        request.resource.data.get('versionLocalEn', '') is string &&
        request.resource.data.get('versionLocalEn', '').size() <= 40 &&
        request.resource.data.estadoCuenta == 'pendiente';
    }

        request.resource.data.uid == resource.data.uid &&
        request.resource.data.email == resource.data.email &&

        request.resource.data.get('versionLocal', 0) >=
          resource.data.get('versionLocal', 0) &&
        request.resource.data.get('versionLocalEn', '') is string &&
        request.resource.data.get('versionLocalEn', '').size() <= 40 &&
        // El alumno puede activar el bloqueo, pero nunca quitar uno existente.
        (
          resource.data.get('pantallaBloqueada', false) != true ||
          request.resource.data.get('pantallaBloqueada', false) == true
        ) &&
        (
          !request.resource.data.diff(resource.data).affectedKeys().hasAny(['seguimientoExtension']) ||
          (
            request.resource.data.seguimientoExtension is map &&
            request.resource.data.seguimientoExtension.keys().hasAll([
              'instalada',
              'activa',
              'version',
              'ultimaSenalEn',
              'dominioActual',
              'tituloActual',
              'visitaIniciadaEn',
              'salidaGrupoId',
              'pendientes',
              'claseId'
            ]) &&
            request.resource.data.seguimientoExtension.keys().hasOnly([
              'instalada',
              'activa',
              'version',
              'ultimaSenalEn',
              'dominioActual',
              'tituloActual',
              'visitaIniciadaEn',
              'salidaGrupoId',
              'pendientes',
              'claseId'
            ]) &&
            request.resource.data.seguimientoExtension.instalada is bool &&
            request.resource.data.seguimientoExtension.activa is bool &&
            request.resource.data.seguimientoExtension.version is string &&
            request.resource.data.seguimientoExtension.version.size() <= 30 &&
            request.resource.data.seguimientoExtension.ultimaSenalEn is string &&
            request.resource.data.seguimientoExtension.ultimaSenalEn.size() <= 40 &&
            request.resource.data.seguimientoExtension.dominioActual is string &&
            request.resource.data.seguimientoExtension.dominioActual.size() <= 120 &&
            request.resource.data.seguimientoExtension.tituloActual is string &&
            request.resource.data.seguimientoExtension.tituloActual.size() <= 300 &&
            request.resource.data.seguimientoExtension.visitaIniciadaEn is string &&
            request.resource.data.seguimientoExtension.visitaIniciadaEn.size() <= 40 &&
            request.resource.data.seguimientoExtension.salidaGrupoId is string &&
            request.resource.data.seguimientoExtension.salidaGrupoId.size() <= 160 &&
            request.resource.data.seguimientoExtension.pendientes is number &&
            request.resource.data.seguimientoExtension.pendientes >= 0 &&
            request.resource.data.seguimientoExtension.pendientes <= 500 &&
            request.resource.data.seguimientoExtension.claseId is string &&
            request.resource.data.seguimientoExtension.claseId.size() <= 120
          )
        ) &&
        (
          !request.resource.data.diff(resource.data).affectedKeys().hasAny(['versionScript']) ||
          (
            request.resource.data.versionScript is string &&
            request.resource.data.versionScript.size() <= 80
          )
        ) &&
        (
          !request.resource.data.diff(resource.data).affectedKeys().hasAny(['presenciaEnLinea']) ||
          request.resource.data.presenciaEnLinea is bool
        ) &&
        (
          !request.resource.data.diff(resource.data).affectedKeys().hasAny(['presenciaActualizadaEn']) ||
          request.resource.data.presenciaActualizadaEn is timestamp
        ) &&
        (
          !request.resource.data.diff(resource.data).affectedKeys().hasAny(['presenciaActualizadaEnCliente']) ||
          (
            request.resource.data.presenciaActualizadaEnCliente is number &&
            request.resource.data.presenciaActualizadaEnCliente > 0
          )
        ) &&
        (
          !request.resource.data.diff(resource.data).affectedKeys().hasAny(['intentosPortapapeles']) ||
          (
            request.resource.data.intentosPortapapeles is map &&
            request.resource.data.intentosPortapapeles.keys().hasAll([
              'total',
              'conteos',
              'riesgo',
              'intentosUltimosDosMinutos',
              'versionScript',
              'ultimoIntento',
              'historial'
            ]) &&
            request.resource.data.intentosPortapapeles.keys().hasOnly([
              'total',
              'conteos',
              'riesgo',
              'intentosUltimosDosMinutos',
              'versionScript',
              'ultimoIntento',
              'historial'
            ]) &&
            request.resource.data.intentosPortapapeles.total is number &&
            request.resource.data.intentosPortapapeles.total >= 0 &&
            request.resource.data.intentosPortapapeles.total <= 100000 &&
            request.resource.data.intentosPortapapeles.conteos is map &&
            request.resource.data.intentosPortapapeles.conteos.keys().hasOnly([
              'copiar',
              'cortar',
              'pegar'
            ]) &&
            request.resource.data.intentosPortapapeles.conteos.copiar is number &&
            request.resource.data.intentosPortapapeles.conteos.cortar is number &&
            request.resource.data.intentosPortapapeles.conteos.pegar is number &&
            request.resource.data.intentosPortapapeles.riesgo in [
              'sin-datos',
              'bajo',
              'medio',
              'alto'
            ] &&
            request.resource.data.intentosPortapapeles.intentosUltimosDosMinutos is number &&
            request.resource.data.intentosPortapapeles.intentosUltimosDosMinutos >= 0 &&
            request.resource.data.intentosPortapapeles.intentosUltimosDosMinutos <= 1000 &&
            request.resource.data.intentosPortapapeles.versionScript is string &&
            request.resource.data.intentosPortapapeles.versionScript.size() <= 80 &&
            request.resource.data.intentosPortapapeles.ultimoIntento is map &&
            request.resource.data.intentosPortapapeles.historial is list &&
            request.resource.data.intentosPortapapeles.historial.size() <= 50
          )
        ) &&
        (
          !request.resource.data.diff(resource.data).affectedKeys().hasAny(['jitsiParticipacion']) ||
          (
            request.resource.data.jitsiParticipacion is map &&
            request.resource.data.jitsiParticipacion.keys().hasAll([
              'llamadaId',
              'tipo',
              'estado',
              'actualizadoEn'
            ]) &&
            request.resource.data.jitsiParticipacion.keys().hasOnly([
              'llamadaId',
              'tipo',
              'estado',
              'actualizadoEn'
            ]) &&
            request.resource.data.jitsiParticipacion.llamadaId is string &&
            request.resource.data.jitsiParticipacion.llamadaId.size() > 0 &&
            request.resource.data.jitsiParticipacion.llamadaId.size() <= 180 &&
            request.resource.data.jitsiParticipacion.tipo in [
              'individual',
              'grupal'
            ] &&
            request.resource.data.jitsiParticipacion.estado in [
              'notificado',
              'unido',
              'rechazado',
              'salio',
              'finalizada'
            ] &&
            request.resource.data.jitsiParticipacion.actualizadoEn ==
              request.time &&
            (
              (
                request.resource.data.jitsiParticipacion.tipo ==
                  'individual' &&
                resource.data.get('jitsiSala', {}).get('id', '') ==
                  request.resource.data.jitsiParticipacion.llamadaId
              ) ||
              (
                request.resource.data.jitsiParticipacion.tipo ==
                  'grupal' &&
                resource.data.get('jitsiGrupal', {}).get('id', '') ==
                  request.resource.data.jitsiParticipacion.llamadaId
              )
            )
          )
        ) &&
        request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(studentMutableFields());
    }

    function validIncompleteStudentRegistration(uid) {
      return ownsDocument(uid) &&
        (
          resource.data.get('uid', '') == '' ||
          resource.data.get('email', '') == '' ||
          !resource.data.keys().hasAny(['estadoCuenta'])
        ) &&
        validNewStudent(uid);
    }

    function validLegacyStudentRecovery(uid) {
      return verifiedEmail() &&
        ownsDocument(uid) &&
        (
          resource.data.get('uid', '') == '' ||
          resource.data.get('email', '') == '' ||
          !resource.data.keys().hasAny(['estadoCuenta'])
        ) &&
        request.resource.data.uid == uid &&
        request.resource.data.email == authenticatedEmail() &&
        request.resource.data.estadoCuenta == 'pendiente' &&
        request.resource.data.get('emailVerificado', false) ==
          request.auth.token.get('email_verified', false) &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'uid',
          'email',
          'estadoCuenta',
          'emailVerificado',
          'nombreGoogle',
          'fotoGoogle',
          'actualizadoEn'
        ]);
    }

    function validStudentResubmission(uid) {
      return verifiedEmail() &&
        ownsDocument(uid) &&
        resource.data.estadoCuenta == 'rechazado' &&
        request.resource.data.estadoCuenta == 'pendiente' &&
        request.resource.data.uid == resource.data.uid &&
        request.resource.data.email == resource.data.email &&
        request.resource.data.emailVerificado ==
          request.auth.token.get('email_verified', false) &&
        request.resource.data.keys().hasAll([
          'estudiante',
          'solicitudAprobacionEn',
          'antecedentesRechazo'
        ]) &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'estudiante',
          'emailVerificado',
          'estadoCuenta',
          'solicitudAprobacionEn',
          'antecedentesRechazo',
          'actualizadoEn'
        ]);
    }

    function validCompatibleMessageStatusUpdate(uid) {
      return verifiedEmail() &&
        ownsDocument(uid) &&
        resource.data.get('mensajeDocenteActual', {}) is map &&
        request.resource.data.get('mensajeDocenteActual', {}) is map &&
        request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['mensajeDocenteActual']) &&
        request.resource.data.mensajeDocenteActual
          .diff(resource.data.mensajeDocenteActual).affectedKeys()
          .hasOnly(['recibido', 'recibidoEn', 'leido', 'leidoEn']) &&
        (
          !request.resource.data.mensajeDocenteActual
            .diff(resource.data.mensajeDocenteActual).affectedKeys()
            .hasAny(['recibido', 'recibidoEn']) ||
          (
            request.resource.data.mensajeDocenteActual.recibido == true &&
            request.resource.data.mensajeDocenteActual.recibidoEn ==
              request.time
          )
        ) &&
        (
          !request.resource.data.mensajeDocenteActual
            .diff(resource.data.mensajeDocenteActual).affectedKeys()
            .hasAny(['leido', 'leidoEn']) ||
          (
            request.resource.data.mensajeDocenteActual.leido == true &&
            request.resource.data.mensajeDocenteActual.leidoEn ==
              request.time &&
            request.resource.data.mensajeDocenteActual.recibido == true
          )
        );
    }

    }

    match /estudiantes/{uid}/historialAccesos/{eventoId} {
      allow get, list: if ownsDocument(uid) || isTeacher();
      allow create: if ownsDocument(uid) &&
        request.resource.data.keys().hasAll([
          'tipo',
          'uid',
          'email',
          'sesionId',
          'versionScript',
          'visibilidad',
          'navegador',
          'sesionesDuplicadas',
          'duracionSesionSegundos',

        request.resource.data.keys().hasOnly([
          'id',
          'tipo',
          'claseId',
          'documentosEliminados',
          'administradorEmail',
          'administradorUid',
          'ejecutadoEn'
        ]) &&
        request.resource.data.id == eventoId &&
        request.resource.data.tipo == 'eliminacion_historial_jitsi' &&
        request.resource.data.claseId is string &&
        request.resource.data.claseId.size() > 0 &&
        request.resource.data.claseId.size() <= 160 &&
        request.resource.data.documentosEliminados is int &&
        request.resource.data.documentosEliminados >= 0 &&
        request.resource.data.administradorEmail == authenticatedEmail() &&
        request.resource.data.administradorUid == request.auth.uid &&
        request.resource.data.ejecutadoEn == request.time;
      allow update, delete: if false;
    }

    match /desafios/{desafioId} {
      allow read: if signedIn();
      allow create, update, delete: if isTeacher();
    }

    // Todo lo no declarado queda denegado.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
